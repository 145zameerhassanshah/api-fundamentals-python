GET:
curl https://jsonplaceholder.typicode.com/posts/1

curl "https://jsonplaceholder.typicode.com/posts?userId=1"

curl -X POST https://jsonplaceholder.typicode.com/posts -H "Content-Type: application/json" -d "{\"title\":\"API Practice\",\"body\":\"Hello from curl\",\"userId\":1}"
